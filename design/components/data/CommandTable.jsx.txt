import React from 'react';

export function CommandTable({ rows }) {
  return React.createElement('table', {
    style: { width: '100%', borderCollapse: 'collapse', fontFamily: 'var(--font-sans)', fontSize: 'var(--text-base)' }
  },
    React.createElement('tbody', null,
      rows.map((r, i) => React.createElement('tr', { key: i, style: { borderBottom: '1px solid var(--border-default)' } },
        React.createElement('td', {
          style: { padding: '16px 28px 16px 0', fontFamily: 'var(--font-mono)', fontSize: 'var(--text-sm)', color: 'var(--accent-check)', whiteSpace: 'nowrap', verticalAlign: 'top' }
        }, r.command),
        React.createElement('td', { style: { padding: '16px 0', color: 'var(--text-secondary)', lineHeight: 'var(--leading-normal)' } }, r.description)
      ))
    )
  );
}
