import React from 'react';

export function Callout({ tone = 'neutral', title, children }) {
  const tones = {
    neutral: { border: 'var(--border-strong)', fg: 'var(--text-primary)' },
    check: { border: 'var(--accent-check)', fg: 'var(--accent-check)' },
    caution: { border: 'var(--status-unclear-fg)', fg: 'var(--status-unclear-fg)' }
  };
  const t = tones[tone] || tones.neutral;
  return React.createElement('div', {
    style: {
      borderLeft: '2px solid ' + t.border, background: 'var(--surface-sunken)',
      borderRadius: '0 var(--radius-sm) var(--radius-sm) 0', padding: '20px 26px', fontFamily: 'var(--font-sans)'
    }
  },
    title ? React.createElement('div', {
      style: { fontWeight: 600, fontSize: 'var(--text-xs)', color: t.fg, marginBottom: '8px', letterSpacing: 'var(--tracking-wider)', textTransform: 'uppercase' }
    }, title) : null,
    React.createElement('div', { style: { fontSize: 'var(--text-base)', color: 'var(--text-secondary)', lineHeight: 'var(--leading-normal)' } }, children)
  );
}
